64const Usuario* SistemaSaveFood::usuarioDaSessaoSemLock(const Sessao& sessao) const {
65 if (!sessao.ativa_) {

76Sessao SistemaSaveFood::entrar(const std::string& email, const std::string& senha) {
77 std::lock_guard<std::mutex> trava(mutex_);
78 if (email.empty() || senha.empty() ||
79 proximoIdSessao_ == std::numeric_limits<int>::max()) {
80 return Sessao();
81 }
82 for (const auto& item : usuarios_) {
83 if (item.second.email() == email && item.second.conferirSenha(senha)) {
84 const int codigo = proximoIdSessao_;
85 if (!sessoesAtivas_.emplace(codigo, item.first).second) {
86 return Sessao();
87 }
88 ++proximoIdSessao_;
89 return Sessao(codigo, item.first);
90 }
91 }
92 return Sessao();
93}

105int SistemaSaveFood::publicarOferta(const Sessao& gerente, const std::string& descricao,
106 int quantidade, double pesoTotalKg, double precoUnitario,
107 const std::string& janelaRetirada,
108 Oferta* ofertaPublicada) {
109 std::lock_guard<std::mutex> trava(mutex_);
110 const Usuario* usuario = usuarioDaSessaoSemLock(gerente);
111 if (!usuario || usuario->perfil() != Perfil::Gerente ||
112 restaurantes_.find(usuario->idRestaurante()) == restaurantes_.end()) {
113 return 0;
114 }
115 // Evita exceder o intervalo dos identificadores inteiros.
116 if (proximoIdOferta_ == std::numeric_limits<int>::max()) {
117 return 0;
118 }
119 const int id = proximoIdOferta_;
120 Oferta oferta(id, usuario->idRestaurante(), descricao, quantidade,
121 pesoTotalKg, precoUnitario, janelaRetirada);
122 if (!oferta.valida()) {
123 return 0;
124 }
125 auto insercao = ofertas_.emplace(id, oferta);
126 if (!insercao.second) {
127 return 0;
128 }
129 try {
130 if (ofertaPublicada) {
131 *ofertaPublicada = oferta;
132 }
133 } catch (...) {
134 // Se a copia do resultado falhar, desfaz a publicacao.
135 ofertas_.erase(insercao.first);
136 throw;
137 }
138 ++proximoIdOferta_;
139 return id;
140}

142std::vector<Oferta> SistemaSaveFood::consultarOfertas(const Sessao& consumidor,
143 const std::string& pesquisa) const {
144 std::lock_guard<std::mutex> trava(mutex_);
145 const Usuario* usuario = usuarioDaSessaoSemLock(consumidor);
146 if (!usuario || usuario->perfil() != Perfil::Consumidor) {
147 return {};
148 }
149 std::vector<Oferta> encontradas;
150 const std::string filtro = paraMinusculas(pesquisa);
151 for (const auto& item : ofertas_) {
152 const Oferta& oferta = item.second;
153 if (oferta.ativa() && oferta.quantidadeDisponivel() > 0 &&
154 paraMinusculas(oferta.descricao()).find(filtro) != std::string::npos) {
155 encontradas.push_back(oferta);
156 }
157 }
158 return encontradas;

161bool SistemaSaveFood::detalhesOferta(const Sessao& sessao, int idOferta,

163 std::lock_guard<std::mutex> trava(mutex_);
164 const Usuario* usuario = usuarioDaSessaoSemLock(sessao);
165 auto it = ofertas_.find(idOferta);

167 return false;

171 usuario->idRestaurante() == encontrada.idRestaurante();
172 const bool consumidor = usuario->perfil() == Perfil::Consumidor &&
173 encontrada.estaAtiva();
174 if (!gerenteDono && !consumidor) {
175 return false;
176 }
177 oferta = encontrada;
178 return true;

181bool SistemaSaveFood::reservar(const Sessao& consumidor, int idOferta, int quantidade,
182 int& idReservaCriada, Reserva* reservaCriada) {
183 std::lock_guard<std::mutex> trava(mutex_);
184 const Usuario* usuario = usuarioDaSessaoSemLock(consumidor);
185 auto itOferta = ofertas_.find(idOferta);
186 if (!usuario || usuario->perfil() != Perfil::Consumidor ||
187 itOferta == ofertas_.end() || !itOferta->second.estaAtiva() ||
188 quantidade <= 0 ||
189 quantidade > itOferta->second.quantidadeDisponivel()) {
190 return false;
191 }
192
193 // A gravacao e a baixa do saldo ficam na mesma regiao critica.
194 if (proximoIdReserva_ == std::numeric_limits<int>::max()) {
195 return false;
196 }
197 const int id = proximoIdReserva_;
198 if (!reservas_.emplace(id, Reserva(id, idOferta, usuario->id(), quantidade)).second) {
199 return false;
200 }
201 if (!itOferta->second.reservar(quantidade)) {
202 reservas_.erase(id);
203 return false;
204 }
205 ++proximoIdReserva_;
206 idReservaCriada = id;
207 if (reservaCriada) {
208 *reservaCriada = reservas_.at(id);
209 }
210 return true;
211}

213bool SistemaSaveFood::cancelarReserva(const Sessao& consumidor, int idReserva) {
214 std::lock_guard<std::mutex> trava(mutex_);
215 const Usuario* usuario = usuarioDaSessaoSemLock(consumidor);
216 auto itReserva = reservas_.find(idReserva);
217 if (!usuario || usuario->perfil() != Perfil::Consumidor ||
218 itReserva == reservas_.end()) {
219 return false;
220 }
221 Reserva& reserva = itReserva->second;
222 if (reserva.idConsumidor() != usuario->id() ||
223 reserva.situacao() != EstadoReserva::Pendente) {
224 return false;
225 }
226 auto itOferta = ofertas_.find(reserva.idOferta());
227 if (itOferta == ofertas_.end()) {
228 return false;
229 }
230
231 // Trabalha com uma copia da reserva: se a restauracao falhar,
232 // nem a situacao original nem o estoque serao alterados.
233 Reserva cancelada = reserva;
234 if (!cancelada.cancelar() ||
235 !itOferta->second.restaurarDisponibilidade(reserva.quantidadeReservada())) {
236 return false;
237 }
238 reserva = cancelada;
239 return true;
240}
